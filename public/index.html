async function ucitajSveIgre() { // 1. Prvo dohvatimo sve kategorije (mape: sportska, logicke, itd.) const resKat = await fetch('/api/kategorije-igara'); const kategorije = await resKat.json(); const listaDiv = document.getElementById('lista-igara'); listaDiv.innerHTML = ""; // 2. Za svaku kategoriju dohvatimo igre unutar nje for (const kat of kategorije) { const resIgre = await fetch(`/api/lista-igara/${kat}`); const igre = await resIgre.json(); igre.forEach(igra => { // Kreiramo link koji zna točnu putanju do podmape const link = document.createElement('a'); link.href = `view-game.html?igra=${igra}&kat=${kat}`; link.innerText = `${igra.replace('.html', '')} (${kat})`; listaDiv.appendChild(link); }); } }
